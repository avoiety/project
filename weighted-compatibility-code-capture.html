의상 간 어울림을 가중치 그래프로 만드는 코드

의상은 정점, 의상 쌍의 호환 점수는 간선의 가중치로 저장합니다.

// 색상·공통 상황·아우터 여부로 두 의상의 호환 가중치를 계산
function compatibilityWeight(first, second) {
  let weight = first.color === second.color ? 5
    : neutralColors.has(first.color) || neutralColors.has(second.color) ? 7
    : complementaryColors.has(`${first.color}-${second.color}`) ? 6 : 2;

  weight += (first.occasions || []).filter((occasion) =>
    (second.occasions || []).includes(occasion)
  ).length * 2;

  return weight + (first.category === "아우터" || second.category === "아우터" ? 1 : 0);
}

// 모든 의상 쌍을 양방향 간선으로 연결한 가중치 그래프 생성
function buildCompatibilityGraph(items) {
  const graph = new Map(items.map((item) => [item.id, new Map()]));
  items.forEach((item, index) => items.slice(index + 1).forEach((other) => {
    const weight = compatibilityWeight(item, other);
    graph.get(item.id).set(other.id, weight);
    graph.get(other.id).set(item.id, weight);
  }));
  return graph;
}
보고서 설명동일색·중립색·보색 여부와 공통 상황 수를 반영해 가중치를 정하고, 각 의상 쌍에 같은 값을 양방향으로 저장하여 무방향 가중치 그래프를 만든다.
